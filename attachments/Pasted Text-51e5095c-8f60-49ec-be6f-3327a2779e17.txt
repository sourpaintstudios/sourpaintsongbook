/* ===================== JS START =======================
   Verbatim from build/app.js — the actual functions that run the
   SPS / STD toggle in the real app. Nothing here is simplified or
   rewritten for this demo; only the two sample songs and the
   render() wiring at the very bottom are demo-specific.

// A "chord-only" line: strip every [Chord] tag out of it and nothing
// but whitespace is left, and there was at least one tag to strip (an
// empty blank line is not a chord line).
function isChordOnlyLine(line) {
  var str = String(line || '');
  if (!/\[([^\[\]]{1,12})\]/.test(str)) return false;
  return str.replace(/\[([^\[\]]{1,12})\]/g, '').trim() === '';
}

function formatChordLine(line) {
  var escaped = String(line || '')
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
  // Wrap bracketed chords like [G] or [C9] in a styled tag for legibility.
  return escaped.replace(/\[([^\[\]]{1,12})\]/g, '<span class="chord">$1</span>');
}

function renderLyricsBody(lyrics, chordLayout) {
  var text = String(lyrics || '').trim();
  if (!text) return '<p class="lyrics-empty">No lyrics yet.</p>';
  var leftLayout = chordLayout === 'left';
  var blocks = text.split(/\n\s*\n/);
  return blocks.map(function (block) {
    var rawLines = block.split(/\r?\n/);
    var html = '';
    for (var i = 0; i < rawLines.length; i++) {
      var line = rawLines[i];
      var next = rawLines[i + 1];
      // Sour Paint Studios layout: a line that's nothing but chord tag(s)
      // — a cue, not something to sing — sits to the left of the lyric
      // line it leads into, instead of stacked above it. Only kicks in
      // when there's an actual lyric line right after it to pair with;
      // anything else (a lone chord line at a block's end, back-to-back
      // chord lines, a line with BOTH a chord and lyric words on it,
      // etc.) falls back to the normal stacked rendering.
      if (leftLayout && isChordOnlyLine(line) && next != null && next.trim() !== '' && !isChordOnlyLine(next)) {
        html += '<div class="lyric-cue-row">' +
          '<span class="lyric-cue-chord">' + formatChordLine(line) + '</span>' +
          '<span class="lyric-cue-line">' + formatChordLine(next) + '</span>' +
        '</div>';
        i++; // consumed the paired lyric line too
      } else {
        html += '<span class="lyric-line">' + formatChordLine(line) + '</span>';
      }
    }
    return '<div class="lyric-section">' + html + '</div>';
  }).join('');
}

// ----- demo-only wiring below: two sample songs, no app state object -----
var SONG_A = "Verse 1\n[G]\nI can't begin to sleep\n[F#]\nThe secrets that I keep\n[B]\nLittle bit of wine\n[C9]\nLittle bit of weed";
var SONG_B = "Verse 1\n[G] I can't begin to sleep\n[F#] The secrets that I keep\n[B] Little bit of wine\n[C9] Little bit of weed";

var layoutState = { a: 'standard', b: 'standard' };

function draw(target) {
  document.getElementById('lyrics-' + target).innerHTML =
    renderLyricsBody(target === 'a' ? SONG_A : SONG_B, layoutState[target]);
  var btn = document.getElementById('toggle-' + target);
  var isLeft = layoutState[target] === 'left';
  btn.textContent = isLeft ? 'SPS' : 'STD';
  btn.classList.toggle('btn-layout-top-active', isLeft);
}

document.getElementById('toggle-a').addEventListener('click', function () {
  layoutState.a = layoutState.a === 'left' ? 'standard' : 'left';
  draw('a');
});
document.getElementById('toggle-b').addEventListener('click', function () {
  layoutState.b = layoutState.b === 'left' ? 'standard' : 'left';
  draw('b');
});
draw('a'); draw('b');
