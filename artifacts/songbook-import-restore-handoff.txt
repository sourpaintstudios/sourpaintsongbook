SOUR PAINT STUDIOS — SONGBOOK
Import + Restore handoff  (copy-paste)
=====================================

What this block does, in the live app:

  Import file  — phone file picker. PDF, Word (.docx), .txt, photos.
  Import image — camera / photo library. Same OCR path.
  Drop zone    — drag a PDF or photo onto Add song.
  Review box   — extracted text always lands here first. "Use this text" saves.
  Backup       — downloads sour-paint-songs.json
  Restore      — pick that JSON and merge songs back in.

PDF.js reads text PDFs. If there's no selectable text, Tesseract.js OCRs the pages.
Mammoth.js reads .docx. Old .doc is rejected (save as .docx or PDF).
Tesseract.js OCRs PNG/JPG/WEBP/HEIC.

Libraries load from CDN on first use:
  https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js
  https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js
  https://cdn.jsdelivr.net/npm/tesseract.js@5/dist/tesseract.min.js
  https://cdnjs.cloudflare.com/ajax/libs/mammoth/1.9.0/mammoth.browser.min.js

This code expects these to already exist in the app:
  byId(id), toast(msg), state.songs, persistNow(), render(),
  editorDraft, openEditorDraft(), saveEditor(afterImport),
  prepareLyricsForLayout(text), isChordOnlyLine(line),
  mergeSongLists(a,b), view, goEditor, plusSvg (optional).

Paste the three sections as-is. Do not rewrite PDF/OCR/Word paths.

  var importQueue = [];
  var ignorePopUntil = 0;
  function armFilePickerGuard() {
    ignorePopUntil = Date.now() + 20000;
  }
  // Android file picker fires a back/popstate. Ignore it while importing
  // or the app dumps to the library and the second file never lands.
  window.addEventListener('popstate', function () {
    if (Date.now() < ignorePopUntil) {
      try { history.pushState({ view: view.name }, ''); } catch (e) {}
      return;
    }
    var review = document.getElementById('import-review-modal');
    if (review && !review.hidden) {
      try { history.pushState({ view: view.name }, ''); } catch (e) {}
      return;
    }
  });
  document.addEventListener('focus', function (e) {
    if (e.target && (e.target.id === 'import-file-input' || e.target.id === 'import-image-input')) {
      armFilePickerGuard();
    }
  }, true);

  function triggerDownload(filename, blob) {
    var a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    setTimeout(function () {
      URL.revokeObjectURL(a.href);
      a.remove();
    }, 1500);
  }

  function exportLibrary(quiet) {
    var payload = {
      v: 1,
      app: 'sour-paint-songbook',
      exportedAt: new Date().toISOString(),
      state: { songs: state.songs, settings: state.settings }
    };
    triggerDownload(
      'sour-paint-songs.json',
      new Blob([JSON.stringify(payload)], { type: 'application/json' })
    );
    if (!quiet) toast('Backup saved to your downloads.');
  }

  function restoreLibraryFile(file) {
    if (!file) return;
    var reader = new FileReader();
    reader.onload = function () {
      try {
        var data = JSON.parse(String(reader.result || ''));
        var incoming = (data && data.state && Array.isArray(data.state.songs)) ? data.state : data;
        if (!incoming || !Array.isArray(incoming.songs) || !incoming.songs.length) {
          toast('No songs in that backup.');
          return;
        }
        state.songs = mergeSongLists(state.songs, incoming.songs);
        if (incoming.settings) state.settings = Object.assign(state.settings, incoming.settings);
        persistNow();
        render();
        toast('Restored ' + incoming.songs.length + ' song' + (incoming.songs.length === 1 ? '' : 's') + '.');
      } catch (err) {
        toast('Could not read that backup.');
      }
    };
    reader.readAsText(file);
  }

  var IMPORT_PDF_JS = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js';
  var IMPORT_PDF_WORKER = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
  var IMPORT_TESSERACT_JS = 'https://cdn.jsdelivr.net/npm/tesseract.js@5/dist/tesseract.min.js';
  var IMPORT_MAMMOTH_JS = 'https://cdnjs.cloudflare.com/ajax/libs/mammoth/1.9.0/mammoth.browser.min.js';
  var importGen = 0;
  var importOcrWorker = null;
  var importOcrBroken = false;
  var importPdfReady = null;
  var importTessReady = null;
  var importMammothReady = null;

  function loadScriptOnce(src) {
    return new Promise(function (resolve, reject) {
      var existing = document.querySelector('script[data-import-src="' + src + '"]');
      if (existing) {
        if (existing.getAttribute('data-loaded') === '1') resolve();
        else existing.addEventListener('load', function () { resolve(); });
        existing.addEventListener('error', function () { reject(new Error('Could not load ' + src)); });
        return;
      }
      var s = document.createElement('script');
      s.src = src;
      s.async = true;
      s.setAttribute('data-import-src', src);
      s.onload = function () { s.setAttribute('data-loaded', '1'); resolve(); };
      s.onerror = function () { reject(new Error('Could not load helper. Check your connection and try again.')); };
      document.head.appendChild(s);
    });
  }

  function ensurePdfJs() {
    if (window.pdfjsLib) {
      window.pdfjsLib.GlobalWorkerOptions.workerSrc = IMPORT_PDF_WORKER;
      return Promise.resolve(window.pdfjsLib);
    }
    if (!importPdfReady) {
      importPdfReady = loadScriptOnce(IMPORT_PDF_JS).then(function () {
        if (!window.pdfjsLib) throw new Error('PDF reader did not start.');
        window.pdfjsLib.GlobalWorkerOptions.workerSrc = IMPORT_PDF_WORKER;
        return window.pdfjsLib;
      });
    }
    return importPdfReady;
  }

  function ensureTesseract() {
    if (window.Tesseract) return Promise.resolve(window.Tesseract);
    if (!importTessReady) importTessReady = loadScriptOnce(IMPORT_TESSERACT_JS).then(function () {
      if (!window.Tesseract) throw new Error('Image reader did not start.');
      return window.Tesseract;
    });
    return importTessReady;
  }

  function ensureMammoth() {
    if (window.mammoth) return Promise.resolve(window.mammoth);
    if (!importMammothReady) {
      importMammothReady = loadScriptOnce(IMPORT_MAMMOTH_JS).then(function () {
        if (!window.mammoth) throw new Error('Word reader did not start.');
        return window.mammoth;
      });
    }
    return importMammothReady;
  }

  function setImportProgress(pct) {
    var bar = byId('import-review-progress');
    var fill = byId('import-review-progress-fill');
    if (!bar || !fill) return;
    if (pct == null) { bar.hidden = true; return; }
    bar.hidden = false;
    fill.style.width = Math.max(0, Math.min(100, pct)) + '%';
  }

  function setImportStatus(msg) {
    var el = byId('import-review-status');
    if (el) el.textContent = msg || '';
  }

  function openImportReviewBusy() {
    var modal = byId('import-review-modal');
    var ta = byId('import-review-text');
    var actions = byId('import-review-actions');
    var busyCancel = byId('import-review-busy-cancel');
    if (ta) { ta.hidden = true; ta.value = ''; }
    if (actions) actions.hidden = true;
    if (busyCancel) busyCancel.hidden = false;
    setImportProgress(8);
    setImportStatus('Reading your file…');
    if (modal) modal.hidden = false;
  }

  function showImportReviewText(text, note) {
    var ta = byId('import-review-text');
    var actions = byId('import-review-actions');
    var busyCancel = byId('import-review-busy-cancel');
    if (ta) {
      ta.hidden = false;
      ta.value = text || '';
    }
    if (actions) actions.hidden = false;
    if (busyCancel) busyCancel.hidden = true;
    setImportProgress(null);
    setImportStatus(note || 'Check line breaks and chords, then use this text.');
  }

  function closeImportReview() {
    importGen += 1;
    var modal = byId('import-review-modal');
    if (modal) modal.hidden = true;
    setImportProgress(null);
  }

  function closeImportMenu() {
    var menu = byId('import-menu');
    if (menu) menu.hidden = true;
  }

  function toggleImportMenu() {
    var menu = byId('import-menu');
    if (!menu) return;
    menu.hidden = !menu.hidden;
  }

  function isPdfFile(file) {
    return file && (file.type === 'application/pdf' || /\.pdf$/i.test(file.name || ''));
  }
  function isImageFile(file) {
    return file && (/^image\//.test(file.type || '') || /\.(png|jpe?g|gif|webp|heic|heif|bmp)$/i.test(file.name || ''));
  }
  function isDocxFile(file) {
    return file && (
      file.type === 'application/vnd.openxmlformats-officedocument.wordprocessingml.document' ||
      /\.docx$/i.test(file.name || '')
    );
  }
  function isOldDocFile(file) {
    return file && (file.type === 'application/msword' || /\.doc$/i.test(file.name || '')) && !isDocxFile(file);
  }
  function isTextFile(file) {
    return file && (/^text\//.test(file.type || '') || /\.(txt|rtf|md|chordpro|cho|crd)$/i.test(file.name || ''));
  }

  function extractDocxText(file) {
    setImportStatus('Opening Word document…');
    return file.arrayBuffer().then(function (buf) {
      return ensureMammoth().then(function (mammoth) {
        return mammoth.extractRawText({ arrayBuffer: buf });
      });
    }).then(function (res) {
      return (res && res.value) ? String(res.value).trim() : '';
    });
  }

  function extractPlainText(file) {
    setImportStatus('Reading text…');
    return new Promise(function (resolve, reject) {
      var reader = new FileReader();
      reader.onload = function () { resolve(String(reader.result || '')); };
      reader.onerror = function () { reject(new Error('Could not read that file.')); };
      reader.readAsText(file);
    });
  }

  function pdfItemsToText(items) {
    var lines = [];
    var line = '';
    var lastY = null;
    for (var i = 0; i < items.length; i++) {
      var item = items[i];
      var str = item.str || '';
      var y = item.transform ? item.transform[5] : 0;
      if (lastY != null && Math.abs(y - lastY) > 3.5) {
        lines.push(line.replace(/[ \t]+$/g, ''));
        line = '';
      }
      line += str;
      if (item.hasEOL) {
        lines.push(line.replace(/[ \t]+$/g, ''));
        line = '';
        lastY = null;
      } else {
        lastY = y;
      }
    }
    if (line) lines.push(line.replace(/[ \t]+$/g, ''));
    return lines.join('\n').replace(/\n{3,}/g, '\n\n').trim();
  }

  function extractPdfText(file, gen) {
    return file.arrayBuffer().then(function (buf) {
      return ensurePdfJs().then(function (pdfjsLib) {
        return pdfjsLib.getDocument({ data: new Uint8Array(buf) }).promise;
      });
    }).then(function (pdf) {
      var maxPages = Math.min(pdf.numPages, 12);
      var pageTexts = [];
      var i = 1;
      function nextPage() {
        if (gen !== importGen) return Promise.reject(new Error('cancelled'));
        if (i > maxPages) {
          return Promise.resolve({ pdf: pdf, text: pageTexts.join('\n\n').trim(), pages: maxPages });
        }
        setImportStatus('Reading PDF page ' + i + ' of ' + maxPages + '…');
        setImportProgress(10 + (i / maxPages) * 40);
        var pageNo = i++;
        return pdf.getPage(pageNo).then(function (page) {
          return page.getTextContent().then(function (tc) {
            pageTexts.push(pdfItemsToText(tc.items || []));
            return nextPage();
          });
        });
      }
      return nextPage();
    });
  }

  function renderPdfPageToDataUrl(page, scale) {
    var viewport = page.getViewport({ scale: scale || 1.6 });
    var canvas = document.createElement('canvas');
    canvas.width = Math.min(1600, viewport.width);
    canvas.height = Math.min(2200, viewport.height);
    var ctx = canvas.getContext('2d');
    var drawVp = page.getViewport({ scale: canvas.width / page.getViewport({ scale: 1 }).width });
    canvas.height = drawVp.height;
    return page.render({ canvasContext: ctx, viewport: drawVp }).promise.then(function () {
      return canvas.toDataURL('image/png');
    });
  }

  function getOcrWorker(gen) {
    return ensureTesseract().then(function (Tesseract) {
      if (importOcrBroken) {
        importOcrWorker = null;
        importOcrBroken = false;
      }
      if (importOcrWorker) return importOcrWorker;
      return Tesseract.createWorker('eng', 1, {
        logger: function (m) {
          if (gen !== importGen) return;
          if (m && m.status === 'recognizing text' && typeof m.progress === 'number') {
            setImportProgress(55 + m.progress * 40);
            setImportStatus('Reading the page… ' + Math.round(m.progress * 100) + '%');
          } else if (m && m.status) {
            setImportStatus('Preparing image reader…');
          }
        }
      }).then(function (w) {
        importOcrWorker = w;
        return w;
      });
    });
  }

  function ocrSource(src, gen) {
    return getOcrWorker(gen).then(function (worker) {
      return worker.recognize(src).then(function (res) {
        return (res && res.data && res.data.text) ? res.data.text : '';
      }).catch(function (err) {
        importOcrBroken = true;
        try { if (importOcrWorker && importOcrWorker.terminate) importOcrWorker.terminate(); } catch (e) {}
        importOcrWorker = null;
        throw err;
      });
    });
  }

  function ocrPdfPages(pdf, pageCount, gen) {
    var n = Math.min(pageCount, 6);
    var out = [];
    var i = 1;
    function next() {
      if (gen !== importGen) return Promise.reject(new Error('cancelled'));
      if (i > n) return Promise.resolve(out.join('\n\n').trim());
      var pageNo = i++;
      setImportStatus('Scanning page ' + pageNo + ' of ' + n + '…');
      setImportProgress(50 + ((pageNo - 1) / n) * 40);
      return pdf.getPage(pageNo).then(function (page) {
        return renderPdfPageToDataUrl(page, 1.6);
      }).then(function (url) {
        return ocrSource(url, gen);
      }).then(function (text) {
        out.push(text);
        return next();
      });
    }
    return next();
  }

  function recycleFileInput(el) {
    if (!el || !el.parentNode) return;
    var neu = el.cloneNode(true);
    neu.value = '';
    el.parentNode.replaceChild(neu, el);
  }

  function ensureDraftForImport() {
    if (editorDraft) return;
    openEditorDraft(null);
    view = { name: 'editor', songId: null };
    render();
  }

  function pickImportFiles(fileList) {
    var files = Array.prototype.slice.call(fileList || []).filter(Boolean);
    if (!files.length) return;
    armFilePickerGuard();
    ensureDraftForImport();
    var first = files.shift();
    importQueue = importQueue.concat(files);
    runImport(first);
  }

  function runImport(file) {
    if (!file) return;
    armFilePickerGuard();
    var gen = ++importGen;
    closeImportMenu();
    openImportReviewBusy();

    var work;
    if (isPdfFile(file)) {
      setImportStatus('Opening PDF…');
      work = extractPdfText(file, gen).then(function (result) {
        if (gen !== importGen) return;
        var text = result.text || '';
        if (text.replace(/\s+/g, '').length >= 8) return text;
        setImportStatus('No selectable text — scanning the pages…');
        return ocrPdfPages(result.pdf, result.pages, gen);
      });
    } else if (isDocxFile(file)) {
      work = extractDocxText(file);
    } else if (isOldDocFile(file)) {
      work = Promise.reject(new Error('That .doc is too old. Save it as .docx or PDF and import that.'));
    } else if (isTextFile(file)) {
      work = extractPlainText(file);
    } else if (isImageFile(file)) {
      setImportStatus('Reading the photo…');
      setImportProgress(20);
      work = ocrSource(file, gen);
    } else {
      work = Promise.reject(new Error('Use a PDF, Word (.docx), text file, or a photo.'));
    }

    work.then(function (text) {
      if (gen !== importGen) return;
      var cleaned = String(text || '').replace(/\u0000/g, '').trim();
      if (!cleaned) {
        showImportReviewText('', 'Nothing readable came through. Try a clearer photo, or paste the lyrics instead.');
        return;
      }
      showImportReviewText(cleaned, 'Check line breaks and chords, then tap Use this text.');
    }).catch(function (err) {
      if (gen !== importGen) return;
      if (err && err.message === 'cancelled') return;
      showImportReviewText('', (err && err.message) ? err.message : 'Could not read that file.');
    });
  }

  function applyImportedText() {
    if (!editorDraft) {
      closeImportReview();
      return;
    }
    var ta = byId('import-review-text');
    var text = ta ? ta.value : '';
    var prepared = prepareLyricsForLayout(text);
    if (!String(prepared || '').trim()) {
      toast('No lyrics in that file. Paste them in, or try a PDF / .docx.');
      return;
    }
    editorDraft.lyrics = prepared;
    if (!String(editorDraft.title || '').trim()) {
      var lines = String(prepared).split(/\r?\n/);
      for (var i = 0; i < lines.length; i++) {
        var line = lines[i].trim();
        if (!line) continue;
        if (isChordOnlyLine(line)) continue;
        editorDraft.title = line.replace(/^\[|\]$/g, '').slice(0, 80);
        break;
      }
    }
    if (!String(editorDraft.title || '').trim()) editorDraft.title = 'Imported song';
    closeImportReview();
    saveEditor('import');
  }

  function wireImportDrop() {
    var drop = byId('import-drop');
    if (!drop || drop.getAttribute('data-wired') === '1') return;
    drop.setAttribute('data-wired', '1');
    drop.addEventListener('dragover', function (e) {
      e.preventDefault();
      drop.classList.add('is-over');
    });
    drop.addEventListener('dragleave', function () { drop.classList.remove('is-over'); });
    drop.addEventListener('drop', function (e) {
      e.preventDefault();
      drop.classList.remove('is-over');
      var f = e.dataTransfer && e.dataTransfer.files && e.dataTransfer.files[0];
      if (f) runImport(f);
    });
  }

  document.addEventListener('change', function (e) {
    var t = e.target;
    if (!t || !t.id) return;
    if (t.id === 'import-file-input' || t.id === 'import-image-input') {
      var list = t.files;
      recycleFileInput(t);
      if (list && list.length) pickImportFiles(list);
    }
    if (t.id === 'restore-library-input') {
      var f = t.files && t.files[0];
      recycleFileInput(t);
      if (f) restoreLibraryFile(f);
    }
  });

  // Click actions to wire in the global click handler:
  //   toggle-import-menu  -> toggleImportMenu()
  //   import-file         -> fileInput.click()
  //   import-image        -> imgInput.click()
  //   import-review-cancel-> closeImportReview()
  //   import-review-use   -> applyImportedText()
  //   export-library      -> exportLibrary()
  // After renderEditor(), call wireImportDrop().
  // After saveEditor from import: if importQueue.length, runImport(importQueue.shift()).
